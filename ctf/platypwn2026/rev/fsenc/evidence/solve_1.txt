import sys, time
from multiprocessing import Pool
from argon2.low_level import hash_secret_raw, Type
magic=bytes.fromhex("4ca36c80fff763eb2be4224f"); SALT=magic+magic[:4]
TARGET=open('flag.bin','rb').read()[12:56]
def check(pw):
    try:
        if hash_secret_raw(pw, SALT, 2, 19456, 1, 44, Type.ID)==TARGET:
            return pw
    except Exception:
        pass
    return None
if __name__=="__main__":
    path=sys.argv[1]; limit=int(sys.argv[2]) if len(sys.argv)>2 else 10**9
    words=[]
    with open(path,'rb') as f:
        for i,line in enumerate(f):
            if i>=limit: break
            w=line.rstrip(b"\r\n")
            if w: words.append(w)
    print("loaded",len(words),"words",flush=True)
    t0=time.time(); found=None
    with Pool(8) as p:
        for i,res in enumerate(p.imap_unordered(check, words, chunksize=64)):
            if res is not None:
                found=res; p.terminate(); break
            if i%20000==0 and i: print(i, "%.0fs"%(time.time()-t0), flush=True)
    print("FOUND:",found)
PY
